npm 包 `@iostore/react` 提供 `useIO` 与 `useIOSelector`，并基于 `useSyncExternalStore` 与 React 渲染模型对齐。

## 基础用法

```tsx
import { io } from '@iostore/store';
import { useIO } from '@iostore/react';

const count = io(0);

export function Counter() {
  const value = useIO(count);
  return <button onClick={() => count.set((v) => v + 1)}>{value}</button>;
}
```

## 派生值

```tsx
import { io } from '@iostore/store';
import { derived } from '@iostore/store/derived';
import { useIO } from '@iostore/react';

const store = io({ count: 0 });
const doubled = derived([store.count], (value) => value * 2);

export function Summary() {
  const value = useIO(doubled);
  return <span>{value}</span>;
}
```

## useIOSelector

```tsx
import { io } from '@iostore/store';
import { useIOSelector } from '@iostore/react';

const store = io({ count: 0, other: 0 });

export function CountOnly() {
  const count = useIOSelector(store, (state) => state.count);
  return <span>{count}</span>;
}
```

如需自定义比较逻辑：

```tsx
const parity = useIOSelector(store, (state) => ({ p: state.count % 2 }), {
  isEqual: (prev, next) => prev.p === next.p,
});
```

## 调度策略

```tsx
const value = useIO(count, { schedule: 'microtask' });
```

可选值：`sync` | `microtask` | `animationFrame`。
