## Quick Start

<Tabs>
  <TabItem label="Vanilla JS">
    ```ts
import { io, batch } from 'io-store';

const store = io({ count: 0, user: { name: 'Ada' } });

store.count.get(); // read
store.user.name.set('Grace'); // deep write

batch(() => {
store.count.update((v) => v + 1);
store.count.update((v) => v + 1);
});

store.subscribeUpdate((update) => {
console.log('Updated:', update);
});

````

  </TabItem>
  <TabItem label="React">
    ```tsx
import { io } from 'io-store';
import { useIO } from 'io-react';

const store = io({ count: 0, user: { name: 'Ada' } });

export default function App() {
  const state = useIO(store);
  const { count, user } = state;

return (
  <div>
    <p>Count: {count}</p>
    <input
      value={user.name}
      onChange={(e) => store.user.name.set(e.target.value)}
    />
    <button onClick={() => store.count.update((v) => v + 1)}>
      Increment
    </button>
  </div>
);
}
````

  </TabItem>
  <TabItem label="Vue">
    ```vue
<script setup lang="ts">
import { io } from 'io-store';
import { useIO } from 'io-vue';

const store = io({ count: 0, user: { name: 'Ada' } });
const state = useIO(store);

const increment = () => store.count.update((v) => v + 1);

</script>

<template>
  <div>
    <p>Count: {{ state.value.count }}</p>
    <input v-model="state.value.user.name" />
    <button @click="increment">Increment</button>
  </div>
</template>
    ```
  </TabItem>
  <TabItem label="Svelte">
    ```svelte
<script>
  import { io } from 'io-store';
  import { toReadable } from 'io-svelte';

const store = io({ count: 0, user: { name: 'Ada' } });
const state = toReadable(store);

</script>

<p>Count: {$state.count}</p>
<input bind:value={$state.user.name} />
<button on:click={() => store.count.update((v) => v + 1)}>
  Increment
</button>
    ```
  </TabItem>
</Tabs>

## Why IO

<CardGrid stagger>
  <Card title="Deep tree updates" icon="open-book">
    Address nested paths directly without changing APIs.
  </Card>
  <Card title="Snapshot safety" icon="approve-check-circle">
    Reads return frozen snapshots for predictable sharing.
  </Card>
  <Card title="Structured updates" icon="add-document">
    Every write is logged as patches you can replay.
  </Card>
  <Card title="Adapters" icon="setting">
    React, Vue, and Svelte adapters share the same contract.
  </Card>
</CardGrid>
