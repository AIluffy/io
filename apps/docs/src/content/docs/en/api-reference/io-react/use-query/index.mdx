## Signature

```ts
<TData, TError = Error>(options: UseQueryOptions<TData, TError>) => UseQueryResult<TData, TError>
```

## Parameters

| Name | Type |
| --- | --- |
| options | `UseQueryOptions<TData, TError>` |

## Notes

- `options.client` is optional.
- If omitted, useQuery uses [getDefaultClient](/en/api-reference/io-query/get-default-client/).
- Pass a custom `client` only for request-scoped SSR, test isolation, or multiple cache domains.

## Returns

`UseQueryResult<TData, TError>`

## Minimal Example

```ts
import { useQuery } from '@iostore/react';

const result = useQuery({
  key: ['todos'],
  queryFn: fetchTodos,
});
```

```ts
import { createQueryClient } from '@iostore/store/query';
import { useQuery } from '@iostore/react';

const client = createQueryClient();
const result = useQuery({
  client,
  key: ['todos'],
  queryFn: fetchTodos,
});
```
