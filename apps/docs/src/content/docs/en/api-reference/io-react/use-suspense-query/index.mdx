## Signature

```ts
<TData, TError = Error>(options: UseQueryOptions<TData, TError>) => UseSuspenseQueryResult<TData, TError>
```

## Minimal Example

```tsx
import { Suspense } from 'react';
import { useSuspenseQuery } from '@iostore/react';

function User() {
  const result = useSuspenseQuery({ key: ['user', 1], queryFn: fetchUser });
  return <div>{result.data.name}</div>;
}
```
