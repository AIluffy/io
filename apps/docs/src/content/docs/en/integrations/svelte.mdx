## toReadable

```ts
import { io } from 'io-store';
import { toReadable } from 'io-svelte';

const count = io(0);
export const countReadable = toReadable(count);
```

## toWritable

```ts
import { toWritable } from 'io-svelte';

export const countWritable = toWritable(count);
```
