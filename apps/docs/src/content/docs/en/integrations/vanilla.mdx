## Read and write

```ts
import { io } from 'io-store';

const state = io({ count: 0 });

state.count();
state.count((v) => v + 1);
```

## Subscribe

```ts
const unsub = state.subscribe((snapshot) => {
  console.log('snapshot', snapshot);
});

const unsubUpdates = state.subscribeUpdate((update) => {
  console.log('patches', update.patches);
});
```

## Use snapshots

```ts
const snapshot = state.snapshot();
```
