## useIo

```ts
import { io } from 'io-store';
import { useIo } from 'io-vue';

const count = io(0);

export function useCounter() {
  const state = useIo(count);
  const inc = () => count((v) => v + 1);
  return { state, inc };
}
```

## ioRef

`ioRef` adapts a Unit into a Vue `ref`.

```ts
import { io } from 'io-store';
import { ioRef } from 'io-vue';

const count = io(0);
const countRef = ioRef(count);
```
