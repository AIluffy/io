## Install

```bash
npm install io-store
```

Add adapters only when needed:

```bash
npm install io-react io-vue io-svelte
```

## Create State

```ts
import { io } from 'io-store';

const store = io({ count: 0, user: { name: 'Ada' } });
```

## Read and Write

```ts
store.count.get();
store.user.name.set('Grace');
store.count.update((v) => v + 1);
```

## Subscribe

```ts
const unsub = store.count.subscribe((value) => {
  console.log('count changed', value);
});

unsub();
```

## Derived Values

```ts
import { derived } from 'io-store';

const doubled = derived(store, (s) => s.count * 2);
```

## Batch Updates

```ts
import { batch } from 'io-store';

batch(() => {
  store.count.update((v) => v + 1);
  store.user.name.set('Lin');
});
```
