## Install

<Steps>

1. Install the core package:

   ```bash
   npm install @iostore/store
   ```

2. Add a framework adapter (optional):

   ```bash
   npm install @iostore/react
   ```

</Steps>

:::note
If your project already uses a framework, install its adapter:
`@iostore/react`, `@iostore/solid`, `@iostore/vue`, or `@iostore/svelte`.
:::

## Create State

```ts
import { io } from '@iostore/store';

const store = io({ count: 0, user: { name: 'Ada' } });
```

## Read and Write

```ts
store.count.get();
store.user.name.set('Grace');
store.count.set((v) => v + 1);
```

## Subscribe

```ts
const unsub = store.count.subscribe((value) => {
  console.log('count changed', value);
});

unsub();
```

## Derived Values

```ts
import { derived } from '@iostore/store';

const doubled = derived(store, (s) => s.count * 2);
```

## Batch Updates

```ts
import { batch } from '@iostore/store';

batch(() => {
  store.count.set((v) => v + 1);
  store.user.name.set('Lin');
});
```

## Use in your framework

<Tabs default="React">
  <TabItem label="React">
    ```tsx
    import { io } from '@iostore/store';
    import { useIO } from '@iostore/react';

    const count = io(0);

    export function Counter() {
      const value = useIO(count);
      return (
        <button onClick={() => count.set((v) => v + 1)}>{value}</button>
      );
    }
    ```

  </TabItem>
  <TabItem label="Vue">
    ```vue
    <script setup lang="ts">
    import { io } from '@iostore/store';
    import { ioRef, useIO } from '@iostore/vue';

    const count = io(0);
    const value = useIO(count);
    const model = ioRef(count);
    const inc = () => count.set((v) => v + 1);
    </script>

    <template>
      <button @click="inc">{{ value }}</button>
      <input v-model="model" type="number" />
    </template>
    ```

  </TabItem>
  <TabItem label="Solid">
    ```tsx
    import { io } from '@iostore/store';
    import { useIO } from '@iostore/solid';

    const count = io(0);

    export function Counter() {
      const value = useIO(count);
      return (
        <button onClick={() => count.set((v) => v + 1)}>{value()}</button>
      );
    }
    ```

  </TabItem>
  <TabItem label="Svelte">
    ```svelte
    <script lang="ts">
      import { io } from '@iostore/store';
      import { toWritable } from '@iostore/svelte';

      const count = io(0);
      const countStore = toWritable(count);
      const inc = () => count.set((v) => v + 1);
    </script>

    <button on:click={inc}>{$countStore}</button>
    ```

  </TabItem>
</Tabs>

## Common gotcha

- Avoid putting the same object instance in an array twice. Create a new object for each item.
