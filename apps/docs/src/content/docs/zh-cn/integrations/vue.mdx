## useOin

```ts
import { oin } from '@oin/store';
import { useOin } from '@oin/vue';

const count = oin(0);

export function useCounter() {
  const state = useOin(count);
  const inc = () => count((v) => v + 1);
  return { state, inc };
}
```

## oinRef

`oinRef` 将 Unit 适配成 Vue `ref`。

```ts
import { oin } from '@oin/store';
import { oinRef } from '@oin/vue';

const count = oin(0);
const countRef = oinRef(count);
```
