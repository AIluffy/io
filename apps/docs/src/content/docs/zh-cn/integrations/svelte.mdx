## toReadable

```ts
import { oin } from '@oin/store';
import { toReadable } from '@oin/svelte';

const count = oin(0);
export const countReadable = toReadable(count);
```

## toWritable

```ts
import { toWritable } from '@oin/svelte';

export const countWritable = toWritable(count);
```
