## useOin

```tsx
import { oin } from '@oin/store';
import { useOin } from '@oin/react';

const count = oin(0);

export function Counter() {
  const value = useOin(count);
  return <button onClick={() => count((v) => v + 1)}>{value}</button>;
}
```

## Derived in React

```tsx
import { derived, oin } from '@oin/store';
import { useOin } from '@oin/react';

const store = oin({ count: 0 });
const doubled = derived([store.count], (value) => value * 2);

export function Summary() {
  const value = useOin(doubled);
  return <span>{value}</span>;
}
```
