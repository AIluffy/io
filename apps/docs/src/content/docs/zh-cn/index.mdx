## 快速开始

<Tabs>
  <TabItem label="纯JavaScript">
  ```ts
  import { io, batch } from 'io-store';

const store = io({ count: 0, user: { name: 'Ada' } });

store.count.get(); // 读取
store.user.name.set('Grace'); // 深层写入

batch(() => {
store.count.update((v) => v + 1);
store.count.update((v) => v + 1);
});

store.subscribeUpdate((update) => {
console.log('Updated:', update);
});

````

  </TabItem>
  <TabItem label="React">
    ```tsx
import { io } from 'io-store';
import { useIO } from 'io-react';

const store = io({ count: 0, user: { name: 'Ada' } });

export default function App() {
  const state = useIO(store);
  const { count, user } = state;

  return (
    <div>
      <p>Count: {count}</p>
      <input
        value={user.name}
        onChange={(e) => store.user.name.set(e.target.value)}
      />
      <button onClick={() => store.count.update((v) => v + 1)}>
        Increment
      </button>
    </div>
  );
}
````

  </TabItem>
  <TabItem label="Vue">
    ```vue
<script setup lang="ts">
import { io } from 'io-store';
import { useIO } from 'io-vue';

const store = io({ count: 0, user: { name: 'Ada' } });
const state = useIO(store);

const increment = () => store.count.update((v) => v + 1);

</script>

<template>
  <div>
    <p>Count: {{ state.value.count }}</p>
    <input v-model="state.value.user.name" />
    <button @click="increment">Increment</button>
  </div>
</template>
    ```
  </TabItem>
  <TabItem label="Svelte">
    ```svelte
<script>
  import { io } from 'io-store';
  import { toReadable } from 'io-svelte';

const store = io({ count: 0, user: { name: 'Ada' } });
const state = toReadable(store);

</script>

<p>Count: {$state.count}</p>
<input bind:value={$state.user.name} />
<button on:click={() => store.count.update((v) => v + 1)}>
  Increment
</button>
    ```
  </TabItem>
</Tabs>

## 为什么选择 IO

<CardGrid stagger>
  <Card title="深层更新" icon="open-book">
    任意深度路径都可以直接读写。

  <Card title="快照安全" icon="approve-check-circle">
