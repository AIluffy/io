## 安装

<Steps>

1. 安装核心包：

   ```bash
   npm install io-store
   ```

2. 安装框架适配（可选）：

   ```bash
   npm install io-react
   ```

</Steps>

## 创建状态

```ts
import { io } from 'io-store';

const count = io(0);

count(); // 读取
count(1); // 写入
count((v) => v + 1); // 更新
```

## 在框架中使用

<Tabs>
  <TabItem label="Core">
    ```ts
    import { io, batch } from 'io-store';

    const store = io({ count: 0, user: { name: 'Ada' } });

    store.count();
    store.user.name('Grace');

    batch(() => {
      store.count((v) => v + 1);
      store.count((v) => v + 1);
    });
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { io } from 'io-store';
    import { useIo } from 'io-react';

    const count = io(0);

    export function Counter() {
      const value = useIo(count);
      return <button onClick={() => count((v) => v + 1)}>{value}</button>;
    }
    ```
  </TabItem>
  <TabItem label="Vue">
    ```ts
    import { io } from 'io-store';
    import { useIo } from 'io-vue';

    const count = io(0);

    export function useCounter() {
      const state = useIo(count);
      const inc = () => count((v) => v + 1);
      return { state, inc };
    }
    ```
  </TabItem>
  <TabItem label="Svelte">
    ```ts
    import { io } from 'io-store';
    import { toWritable } from 'io-svelte';

    const count = io(0);
    export const countStore = toWritable(count);
    ```
  </TabItem>
</Tabs>
